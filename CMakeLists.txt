cmake_minimum_required(VERSION 3.16)
project(STM32GPIOConfigurator VERSION 3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
qt_add_executable(STM32GPIOConfigurator src/main.cpp src/mainwindow.cpp include/mainwindow.h)
target_include_directories(STM32GPIOConfigurator PRIVATE include)
target_compile_definitions(STM32GPIOConfigurator PRIVATE GPIO_TEMPLATE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/templates")
target_compile_options(STM32GPIOConfigurator PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(STM32GPIOConfigurator PRIVATE Qt6::Widgets)
